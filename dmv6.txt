
# EXPERIMENT: DATA AGGREGATION - SALES PERFORMANCE ANALYSIS

import pandas as pd
import matplotlib.pyplot as plt


# 1. IMPORT THE SALES DATASET
# Load the customer shopping data from a CSV file

df = pd.read_csv("customer_shopping_data.csv")


# 2. EXPLORE THE DATASET
# Display the first and last records and understand the dataset structure

print(df.head())
print(df.tail())
print(df.info())


# 3. IDENTIFY RELEVANT VARIABLES AND COUNT RECORDS
# Identify shopping mall as region/branch, price as sales amount
# and category as product category

# Count the number of records grouped by shopping mall
print(df.groupby("shopping_mall").count())

# Count the number of records grouped by product category
print(df.groupby("category").count())


# 4. CALCULATE TOTAL SALES BY BRANCH AND PRODUCT CATEGORY
# Use groupby() and sum() to calculate the total sales amount
# for each shopping mall and each product category

# Calculate total sales for each branch
branch_sales = df.groupby("shopping_mall")["price"].sum()
print("Total Sales by Branch:")
print(branch_sales)

# Calculate total sales for each product category
category_sales = df.groupby("category")["price"].sum()
print("Total Sales by Category:")
print(category_sales)


# 5. CREATE PIE CHARTS TO VISUALIZE SALES DISTRIBUTION
# Display the percentage contribution of each shopping mall
# and product category to total sales

# Pie chart for sales by branch
plt.figure(figsize=(8, 6))
plt.pie(branch_sales, labels=branch_sales.index, autopct="%1.1f%%")
plt.title("Sales Distribution by Shopping Mall")
plt.show()

# Pie chart for sales by product category
plt.figure(figsize=(8, 6))
plt.pie(category_sales, labels=category_sales.index, autopct="%1.1f%%")
plt.title("Sales Distribution by Product Category")
plt.show()


# 6. IDENTIFY TOP-PERFORMING BRANCHES AND PRODUCT CATEGORIES
# Sort total sales in descending order to identify
# the branches and categories generating the highest sales

# Identify top-performing branches
print("Top Performing Branches:")
print(branch_sales.sort_values(ascending=False))

# Identify top-selling categories
print("Top Selling Categories:")
print(category_sales.sort_values(ascending=False))


# 7. AGGREGATE SALES BY BRANCH AND PRODUCT CATEGORY
# Calculate total sales for every combination of shopping mall
# and product category using groupby()

combined_branch_category_sales = df.groupby(
    ["shopping_mall", "category"]
)["price"].sum()

print("Sales by Branch and Product Category:")
print(combined_branch_category_sales)


# 8. CREATE A PIVOT TABLE AND GROUPED BAR CHART
# Create a summary table to compare product category sales
# across different shopping malls and visualize the comparison

combined_pivot = df.pivot_table(
    index="shopping_mall",
    columns="category",
    values="price",
    aggfunc="sum"
)

print(combined_pivot)

# Create a grouped bar chart to compare sales across shopping malls
combined_pivot.plot(kind="bar", figsize=(12, 6))
plt.title("Sales by Product Category Across Shopping Malls")
plt.xlabel("Shopping Mall")
plt.ylabel("Total Sales Amount")
plt.xticks(rotation=45)
plt.legend(title="Product Category")
plt.tight_layout()
plt.show()
