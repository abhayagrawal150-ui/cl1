import pandas as pd  # For data manipulation
import numpy as np  # For numerical operations
import matplotlib.pyplot as plt  # For plotting graphs
from sklearn.preprocessing import StandardScaler  # For feature standardization
from sklearn.decomposition import PCA  # For dimensionality reduction


df = pd.read_csv('wine.csv')  # Load the dataset
X = df.drop(columns=['Customer_Segment'])  # Select input features
y = df['Customer_Segment']  # Select target variable


plt.figure(figsize=(10,6))  # Set figure size

# Plot data for each customer segment
plt.scatter(X[y==1]['Alcohol'], X[y==1]['Malic_Acid'], label='Customer Segment 1', alpha=0.7)
plt.scatter(X[y==2]['Alcohol'], X[y==2]['Malic_Acid'], label='Customer Segment 2', alpha=0.7)
plt.scatter(X[y==3]['Alcohol'], X[y==3]['Malic_Acid'], label='Customer Segment 3', alpha=0.7)

plt.xlabel('Alcohol')  # Set X-axis label
plt.ylabel('Malic Acid')  # Set Y-axis label
plt.legend()  # Display legend
plt.title('Scatter Plot (Original Data)')  # Set plot title
plt.show()  # Display the plot



scaler = StandardScaler()  # Create StandardScaler object
X_scaled = scaler.fit_transform(X)  # Standardize the features

pca = PCA(n_components=2)  # Create PCA object with 2 components
X_pca = pca.fit_transform(X_scaled)  # Apply PCA to reduce dimensions


pca_df = pd.DataFrame(data=X_pca, columns=['Principal Component 1', 'Principal Component 2'])  # Create DataFrame with PCA components
pca_df['Customer_Segment'] = y  # Add target column


plt.figure(figsize=(10,6))  # Set figure size

# Plot each customer segment using PCA components
plt.scatter(pca_df[pca_df['Customer_Segment']==1]['Principal Component 1'], pca_df[pca_df['Customer_Segment']==1]['Principal Component 2'], label='Customer Segment 1', alpha=0.7)
plt.scatter(pca_df[pca_df['Customer_Segment']==2]['Principal Component 1'], pca_df[pca_df['Customer_Segment']==2]['Principal Component 2'], label='Customer Segment 2', alpha=0.7)
plt.scatter(pca_df[pca_df['Customer_Segment']==3]['Principal Component 1'], pca_df[pca_df['Customer_Segment']==3]['Principal Component 2'], label='Customer Segment 3', alpha=0.7)

plt.xlabel('Principal Component 1')  # Set X-axis label
plt.ylabel('Principal Component 2')  # Set Y-axis label
plt.legend()  # Display legend
plt.title('PCA: Wine dataset')  # Set plot title
plt.show()  # Display the plot

print("Explained variance ratio:", pca.explained_variance_ratio_)  # Display variance explained by each component
print("Total variance captured:", sum(pca.explained_variance_ratio_)*100, "%")  # Display total variance captured
