import pandas as pd  # For data manipulation
import numpy as np  # For numerical operations
import matplotlib.pyplot as plt  # For plotting graphs

from sklearn.cluster import KMeans  # For K-Means clustering
from sklearn.preprocessing import StandardScaler  # For standardizing features

# Load the Iris dataset
df = pd.read_csv("iris.csv")  # Read dataset
print(df.head())  # Display first 5 rows

# Select features (exclude Id and Species)
X = df.iloc[:, 1:-1]  # Select input features

# Standardize the data
scaler = StandardScaler()  # Create scaler
X_scaled = scaler.fit_transform(X)  # Standardize features

# Determine optimal number of clusters using Elbow Method
inertia = []  # Store inertia values

for k in range(1, 11):
    kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)  # Create K-Means model
    kmeans.fit(X_scaled)  # Fit model
    inertia.append(kmeans.inertia_)  # Store inertia

print("Inertia:", inertia)

# Plot Elbow Curve
plt.figure(figsize=(8, 6))  # Set figure size
plt.plot(range(1, 11), inertia, marker='o', linestyle='--')  # Plot inertia values
plt.xlabel("Number of Clusters")  # Set X-axis label
plt.ylabel("Inertia")  # Set Y-axis label
plt.title("Elbow Method")  # Set plot title
plt.grid()  # Display grid
plt.show()  # Show plot

# Apply K-Means with optimal K = 3
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)  # Create model with 3 clusters
y_pred = kmeans.fit_predict(X_scaled)  # Fit model and predict clusters

# Display cluster assignments
df['Cluster'] = y_pred  # Add cluster labels to dataset

print("\nCluster Assignments:")
print(df[['Cluster']].head(10))  # Display first 10 cluster assignments

print("\nCluster Centers:")
print(kmeans.cluster_centers_)  # Display cluster centers
