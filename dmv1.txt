
import numpy as np
import pandas as pd
import json
import matplotlib.pyplot as plt

# 1. LOAD DATA FROM DIFFERENT FILE FORMATS (CSV, EXCEL AND JSON)

# Load CSV and Excel files into Pandas DataFrames
csv = pd.read_csv("sales_data_sample.csv", encoding="cp1252")
ed = pd.read_excel("Sample-Sales-Data.xlsx")

# Load JSON data and convert it into a Pandas DataFrame
with open("customers.json", "r") as file:
    json_data = json.load(file)

json_df = pd.json_normalize(json_data)


# 2. EXPLORE THE DATASETS TO UNDERSTAND THEIR STRUCTURE AND CONTENT

# Display the first few records, dataset information and descriptive statistics
print("CSV Data")
print(csv.head())
csv.info()
print(csv.describe())

print("\nExcel Data")
print(ed.head())
ed.info()
print(ed.describe())

print("\nJSON Data")
print(json_df.head())
json_df.info()


# 3. DATA CLEANING - REMOVE DUPLICATES AND HANDLE MISSING VALUES

# Remove duplicate records from all three datasets
csv = csv.drop_duplicates()
ed = ed.drop_duplicates()
json_df = json_df.drop_duplicates()

# Remove rows containing missing values in important columns
csv = csv.dropna(subset=['SALES', 'ORDERNUMBER', 'PRODUCTLINE'])
ed = ed.dropna()
json_df = json_df.dropna()


# 4. UNIFIED FORMAT - CHECK CLEANED DATASET DIMENSIONS

# Keep separate DataFrames because their structures differ
# Display the number of rows and columns after cleaning

print("\nCleaned CSV:", csv.shape)
print("Cleaned Excel:", ed.shape)
print("Cleaned JSON:", json_df.shape)


# 5. DATA TRANSFORMATION - CREATE NEW COLUMNS AND CATEGORIES

# CSV: Categorize sales amounts into Low, Medium and High
csv['Sales_Category'] = pd.cut(
    csv['SALES'],
    bins=[0, 1000, 5000, float('inf')],
    labels=['Low', 'Medium', 'High']
)

# Excel: Categorize sales representative performance based on sales value
ed['Value_Category'] = pd.cut(
    ed['Value'],
    bins=[0, 30000, 70000, float('inf')],
    labels=['Low', 'Medium', 'High']
)

# JSON: Combine first name and last name to create a Full_Name column
json_df['Full_Name'] = (
    json_df['first'] + ' ' + json_df['last']
)


# 6. DATA ANALYSIS - CALCULATE SALES METRICS AND AGGREGATE DATA

# Calculate total sales, average sales and average order value
total_sales = csv['SALES'].sum()
print("\nTotal Sales:", total_sales)

average_sales = csv['SALES'].mean()
print("Average Sales:", average_sales)

average_order_value = csv.groupby(
    'ORDERNUMBER'
)['SALES'].sum().mean()

print("Average Order Value:", average_order_value)

# Calculate total sales for each product category
category_sales = csv.groupby(
    'PRODUCTLINE'
)['SALES'].sum()

print("\nSales by Product Category:")
print(category_sales)

# Calculate total sales generated by each sales representative
rep_sales = ed.groupby(
    'Sales_Rep_Name'
)['Value'].sum()

print("\nSales by Representative:")
print(rep_sales)

# Count the number of customers belonging to each country
customer_country = json_df['country'].value_counts()

print("\nCustomers by Country:")
print(customer_country.head())


# 7. DATA VISUALIZATION - REPRESENT SALES DATA USING DIFFERENT PLOTS

# Bar Plot: Visualize total sales across different product categories
category_sales.plot(kind='bar')
plt.title('Sales by Product Category')
plt.xlabel('Product Category')
plt.ylabel('Total Sales')
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()

# Pie Chart: Visualize the percentage contribution of each sales representative
rep_sales.plot(kind='pie', autopct='%1.1f%%')
plt.title('Sales by Representative')
plt.ylabel('')
plt.show()

# Box Plot: Visualize the distribution of sales and identify potential outliers
csv['SALES'].plot(kind='box')
plt.title('Sales Distribution')
plt.ylabel('Sales')
plt.show()
